{{- /* Static SVG rate chart, no JS. Input: state ctx. With 3+ monthly points in data/eia_series.yaml: line chart, state vs U.S. Otherwise: bars for the two sourced months in data/eia_prices.yaml plus the U.S. average. */ -}} {{- $c := . -}} {{- $name := $c.d.name -}} {{- $ser := site.Data.eia_series -}} {{- $vals := slice -}} {{- with $ser }}{{ with index .states $c.key }}{{ $vals = . }}{{ end }}{{ end -}} {{- if ge (len $vals) 3 -}} {{- $months := $ser.months -}} {{- $us := $ser.us -}} {{- $all := $vals -}}{{ if eq (len $us) (len $vals) }}{{ $all = $all | append $us }}{{ end -}} {{- $nums := slice }}{{ range $all }}{{ $nums = $nums | append (float .) }}{{ end -}} {{- $lo := math.Floor (sub (index (sort $nums) 0) 1) -}} {{- $hi := math.Ceil (add (index (sort $nums "value" "desc") 0) 1) -}} {{- $w := 640.0 }}{{ $h := 220.0 }}{{ $l := 44.0 }}{{ $r := 12.0 }}{{ $t := 14.0 }}{{ $b := 30.0 -}} {{- $n := sub (len $vals) 1 -}} {{- $sx := div (sub (sub $w $l) $r) (float $n) -}} {{- $sy := div (sub (sub $h $t) $b) (sub $hi $lo) -}} {{- $pts := slice }}{{ range $i, $v := $vals }}{{ $pts = $pts | append (printf "%.1f,%.1f" (add $l (mul $sx (float $i))) (sub (sub $h $b) (mul $sy (sub (float $v) $lo)))) }}{{ end -}} {{- $upts := slice }}{{ if eq (len $us) (len $vals) }}{{ range $i, $v := $us }}{{ $upts = $upts | append (printf "%.1f,%.1f" (add $l (mul $sx (float $i))) (sub (sub $h $b) (mul $sy (sub (float $v) $lo)))) }}{{ end }}{{ end -}} {{- else -}}{{- with $c.p -}} {{- $us := float $c.e.us.y2026 -}} {{- $rows := slice (dict "label" $c.e.prev_month "v" (float .y2025) "cls" "rb-prev") (dict "label" $c.e.month "v" (float .y2026) "cls" "rb-now") (dict "label" (printf "U.S., %s" $c.e.month) "v" $us "cls" "rb-us") -}} {{- $max := 0.0 }}{{ range $rows }}{{ if gt .v $max }}{{ $max = .v }}{{ end }}{{ end -}} {{- end -}}{{- end -}}
